(function(){

     Ultima actualizacion: 18/08/2026                            */
  var DX = {
    'US': { now:'$16,99',      old:'$67,99',       cur:'USD' },
    'CL': { now:'$15.990',     old:'$63.990',      cur:'CLP' },

  var EU = ["AD","AT","BE","BG","CY","DE","EE","ES","FI","FR","GR","HR",
            "IE","IT","LT","LU","LV","MC","ME","MT","NL","PT","SI","SK",
            "SM","VA","XK"];

  /* Zonas horarias -> market. Fallback instantaneo y sin red:
     evita el parpadeo a USD y funciona con bloqueadores activos. */
  var TZ = {
    'America/Santiago':'CL', 'America/Punta_Arenas':'CL',
    'America/Bogota':'CO', 'America/Lima':'PE', 'America/Montevideo':'UY',
    'America/Mexico_City':'MX', 'America/Cancun':'MX', 'America/Merida':'MX',
    'America/Monterrey':'MX', 'America/Matamoros':'MX', 'America/Chihuahua':'MX',

    "Europe/Lisbon","Europe/Amsterdam","Europe/Brussels","Europe/Vienna",
    "Europe/Dublin","Europe/Athens","Europe/Helsinki","Europe/Tallinn",

  function resolve(cc){
    cc = (cc || '').toString().trim().toUpperCase();
    if(cc === 'EU') return 'EU';
    if(EU.indexOf(cc) !== -1) return 'EU';
    return Object.prototype.hasOwnProperty.call(DX, cc) ? cc : 'DEFAULT';
  }

  function guessByTimezone(){
    try{
      var tz = Intl.DateTimeFormat().resolvedOptions().timeZone || '';
      if(TZ[tz]) return TZ[tz];
      if(TZ_EU.indexOf(tz) !== -1) return 'EU';
      if(tz.indexOf('America/Argentina/') === 0) return 'AR';
      if(tz === 'America/Buenos_Aires' || tz === 'America/Cordoba' ||
         tz === 'America/Mendoza' || tz === 'America/Rosario') return 'AR';
    }catch(e){}
    return null;
  }

  function apply(cc){
    var mkt = resolve(cc);
    if(mkt === current) return;      /* evita repintar de mas */

    var p = DX[mkt];

    root.querySelectorAll('.dx-buybar-old').forEach(function(el){ el.textContent = p.old; });
  }

  var forced = null;

  if(forced){ apply(forced); remember(resolve(forced)); return; }

  /* 2) Cache de sesion: el visitante que ya fue geolocalizado no vuelve
        a ver el parpadeo ni gasta otra llamada a la API. */
  var cached = null;
  try { cached = sessionStorage.getItem('dxmkt'); } catch(e){}
  if(cached){ apply(cached); }
  /* 3) Estimacion instantanea por huso horario (sin red) */
  else {
    var guess = guessByTimezone();
    if(guess) apply(guess);
  }

  /* 4) Confirmacion por IP, con timeout: si la API tarda o esta

  var ctrl = null, timer = null;

    if(window.AbortController){
      ctrl = new AbortController();
      timer = setTimeout(function(){ ctrl.abort(); }, 3000);
    }

  fetch('https://get.geojs.io/v1/ip/country.json', ctrl ? {signal:ctrl.signal} : undefined)
    .then(function(r){ if(!r.ok) throw new Error('geo ' + r.status); return r.json(); })
    .then(function(d){
      if(timer) clearTimeout(timer);
      var cc = d && (d.country || d.country_code);
      if(!cc) return;
      apply(cc);
      remember(resolve(cc));
    })
    .catch(function(){ if(timer) clearTimeout(timer); });
})();
